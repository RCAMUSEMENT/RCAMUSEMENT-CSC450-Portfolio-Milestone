START Application
    // Encapsulate structural state within a cache-line aligned telemetry block
    STRUCTURE SubmersibleTelemetry
        INITIALIZE depth_pressure_psi = 0
        INITIALIZE hull_mutex
        INITIALIZE ascent_descent_barrier
        INITIALIZE diving_phase_complete = false
    END STRUCTURE

    FUNCTION execute_ballast_dive(telemetry_reference)
        FOR psi_index = 1 TO 20
            LOCK telemetry_reference.hull_mutex
            SET telemetry_reference.depth_pressure_psi = psi_index

            // Atomic console telemetry logging isolation
            LOCK console_io_mutex
                PRINT "[Telemetry Thread 1] Ballast Intake Active - Hull Hydrostatic Pressure: " + telemetry_reference.depth_pressure_psi + " PSI"
            UNLOCK console_io_mutex

            UNLOCK telemetry_reference.hull_mutex
        ENDFOR

        LOCK telemetry_reference.hull_mutex
        SET telemetry_reference.diving_phase_complete = true
        SIGNAL telemetry_reference.ascent_descent_barrier
        UNLOCK telemetry_reference.hull_mutex
    END FUNCTION

    FUNCTION execute_ballast_ascent(telemetry_reference)
        LOCK telemetry_reference.hull_mutex
        // Evaluate predicate condition loop to explicitly prevent spurious wake-ups
        WHILE telemetry_reference.diving_phase_complete IS false
            WAIT on telemetry_reference.ascent_descent_barrier WITH telemetry_reference.hull_mutex
        ENDWHILE
        UNLOCK telemetry_reference.hull_mutex

        WHILE true
            LOCK telemetry_reference.hull_mutex
            IF telemetry_reference.depth_pressure_psi < 0 THEN
                UNLOCK telemetry_reference.hull_mutex
                BREAK loop
            ENDIF

            INITIALIZE current_pressure = telemetry_reference.depth_pressure_psi
            DECREMENT telemetry_reference.depth_pressure_psi
            UNLOCK telemetry_reference.hull_mutex

            LOCK console_io_mutex
                PRINT "[Telemetry Thread 2] Ascent Initiated - Blowing Ballast - Hull Hydrostatic Pressure: " + current_pressure + " PSI"
            UNLOCK console_io_mutex
        ENDWHILE
    END FUNCTION

    INITIALIZE global_submersible_telemetry
    SPAWN Dive_Worker passing global_submersible_telemetry
    SPAWN Ascent_Worker passing global_submersible_telemetry

    JOIN Dive_Worker to main processing thread
    JOIN Ascent_Worker to main processing thread
END Application
